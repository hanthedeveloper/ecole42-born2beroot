root@haincel42:~# cat /usr/local/bin/monitoring.sh
#!/bin/bash

export LC_ALL=C

arch=$(uname -a)

cpu_phys=$(grep "physical id" /proc/cpuinfo | sort -u | wc -l)
[ "$cpu_phys" -eq 0 ] && cpu_phys=$(lscpu | awk '/^Socket\(s\):/ {print $2}')
cpu_virt=$(grep -c "^processor" /proc/cpuinfo)

ram_used=$(free -m | awk '/Mem:/ {print $3}')
ram_total=$(free -m | awk '/Mem:/ {print $2}')
ram_pct=$(free -m | awk '/Mem:/ {printf("%.2f", $3/$2*100)}')

disk_used=$(df -m | awk '/^\/dev\// {u+=$3} END {print u}')
disk_total=$(df -m | awk '/^\/dev\// {t+=$2} END {printf("%.1f", t/1024)}')
disk_pct=$(df -m | awk '/^\/dev\// {u+=$3; t+=$2} END {printf("%d", u/t*100)}')

cpu_load=$(vmstat 1 2 | tail -1 | awk '{printf("%.1f", 100-$15)}')

last_boot=$(uptime -s | cut -c1-16)

if [ "$(lsblk | grep -c lvm)" -gt 0 ]; then lvm="yes"; else lvm="no"; fi

tcp=$(ss -Htn state established | wc -l)
users=$(who | wc -l)

ip=$(hostname -I | awk '{print $1}')
mac=$(ip link | awk '/link\/ether/ {print $2; exit}')

sudo_cmd=$(grep -c "COMMAND" /var/log/sudo/sudo_config)

wall << EOF
        #Architecture: $arch
        #Physical CPU: $cpu_phys
        #vCPU: $cpu_virt
        #Memory Usage: $ram_used/${ram_total}MB ($ram_pct%)
        #Disk Usage: $disk_used/${disk_total}Gb ($disk_pct%)
        #CPU load: $cpu_load%
        #Last boot: $last_boot
        #LVM use: $lvm
        #TCP Connections: $tcp ESTABLISHED
        #User log: $users
        #Network: IP $ip ($mac)
        #Sudo: $sudo_cmd cmd
EOF

Bu satır, VM'inde şu an kurulu bir SSH bağlantısı olduğunu gösteriyor. Sütun sütun:
Netid: tcp: TCP bağlantısı.
State: ESTAB: Established, yani bağlantı kurulmuş ve aktif. (Daha önce gördüğün LISTEN
"bekliyorum" demekti, ESTAB "biri bağlandı, konuşuluyor" demek.)
Recv-Q / Send-Q: 0 / 0: Bekleyen veri yok, bağlantı sakin.
Local Address:Port: 10.0.2.15:4242: Bağlantının VM tarafı. VM'in IP'si ve SSH portu.
Peer Address:Port: 10.0.2.2:44308: Bağlanan taraf. 10.0.2.2 VirtualBox NAT'te host makinenin
(kendi bilgisayarının) VM'den görünen adresi. 44308 ise host tarafının rastgele seçtiği geçici port.
